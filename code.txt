6. src/components/Sidebar.jsx
export default function Sidebar({ chapters, active, setActive }) {
  return (
    <div className="w-80 h-screen overflow-y-auto bg-slate-900 border-r border-slate-800 p-4">
      <h1 className="text-2xl font-bold mb-6">
        AI Creator OS
      </h1>

      <div className="space-y-3">
        {chapters.map((chapter) => (
          <button
            key={chapter.id}
            onClick={() => setActive(chapter)}
            className={`w-full text-left p-4 rounded-xl transition-all ${
              active.id === chapter.id
                ? "bg-blue-600"
                : "bg-slate-800 hover:bg-slate-700"
            }`}
          >
            <h2 className="font-semibold">
              {chapter.id}. {chapter.title}
            </h2>

            <p className="text-sm text-slate-300 mt-1">
              {chapter.description}
            </p>
          </button>
        ))}
      </div>
    </div>
  );
}
7. src/components/PromptBlock.jsx
export default function PromptBlock({ prompt }) {
  const copyPrompt = () => {
    navigator.clipboard.writeText(prompt);
  };

  return (
    <div className="bg-black rounded-xl p-4 relative mt-3">
      <button
        onClick={copyPrompt}
        className="absolute top-3 right-3 bg-blue-600 px-3 py-1 rounded-lg text-sm"
      >
        Copy
      </button>

      <pre className="whitespace-pre-wrap text-green-400">
        {prompt}
      </pre>
    </div>
  );
}
8. src/components/SectionCard.jsx
import PromptBlock from "./PromptBlock";

export default function SectionCard({ section }) {
  return (
    <div className="bg-slate-900 p-6 rounded-2xl mb-6 border border-slate-800">
      <h2 className="text-2xl font-bold mb-4">
        {section.heading}
      </h2>

      {section.content &&
        (Array.isArray(section.content) ? (
          <ul className="list-disc ml-5 space-y-2 text-slate-300">
            {section.content.map((item, index) => (
              <li key={index}>{item}</li>
            ))}
          </ul>
        ) : (
          <p className="text-slate-300 leading-8">
            {section.content}
          </p>
        ))}

      {section.prompt && (
        <PromptBlock prompt={section.prompt} />
      )}

      {section.bestFor && (
        <div className="mt-5">
          <h3 className="font-semibold mb-2">
            Best For:
          </h3>

          <ul className="list-disc ml-5 text-slate-300">
            {section.bestFor.map((item, i) => (
              <li key={i}>{item}</li>
            ))}
          </ul>
        </div>
      )}

      {section.bestContent && (
        <div className="mt-5">
          <h3 className="font-semibold mb-2">
            Best Content:
          </h3>

          <ul className="list-disc ml-5 text-slate-300">
            {section.bestContent.map((item, i) => (
              <li key={i}>{item}</li>
            ))}
          </ul>
        </div>
      )}
    </div>
  );
}
